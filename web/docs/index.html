🐡 What is Unfish?

Unfish is a real, general-purpose programming language, runtime, compiler, and educational computing environment written in clean ANSI C99 with zero external dependencies.

It is designed to provide an unbroken path of intellectual ascent: from visual block programming, into clean indentation-based text programming, to algorithms, data structures, runtime architecture, garbage collection, bytecode virtual machines, and low-level systems programming.

"Hide complexity until the learner is ready — never hide it forever."

Five Execution Engines in Lockstep

Unfish implements five distinct execution backends, kept in lockstep by a differential test suite that requires byte-identical output from every one of them:

  1. Tree-Walking AST Interpreter: Immediate startup, direct AST evaluation, ideal for REPL and visual block editing.
  2. Bytecode Virtual Machine (--vm): 57-opcode stack VM with lexical upvalue capture cells, exception unwinding, and 2x–6x speedup over AST tree-walking.
  3. Register-Based Bytecode VM (--regvm): 256-register, 3-address VM with computed-goto dispatch.
  4. Native C99 AOT Compiler (build): Direct transpilation into standalone ANSI C99 with single-header runtime, measured at 25–60x faster than AST.
  5. WebAssembly Target (build --wasm): Portable WASM binary running in browsers and edge runtimes.

15. Memory Model & Garbage Collector

        

          The Unfish runtime is designed for predictable low latency and tiny memory footprints:
        

        
    
          
  • Tagged Unions (16 bytes): All stack values fit inside UfValue containing a 64-bit payload and an 8-bit type tag.
    • 
          
  • Arena Allocators: Parsing AST nodes and compilation bytecode chunks use fast linear bump arenas freed in O(1) time.
    • 
          
  • Mark-Sweep GC: Objects on the heap (strings, arrays, maps, instances, fibers) are managed by an exact tri-color mark-sweep garbage collector.
    •

📜 Formal Grammar & Language Specification

        

          Unfish syntax is formally specified in Extended Backus-Naur Form (ISO/IEC 14977). The language utilizes lexical indentation (the off-side rule), structured expression pipelines, algebraic data types, and pattern matching.
        


        

          Lexical Note: Indentation in Unfish is significant. The lexer tracks leading whitespace on non-blank lines using an internal indentation stack. An increase in indentation emits a virtual TOKEN_INDENT, while a decrease emits one or more TOKEN_DEDENT tokens. Newlines within parentheses (...), brackets [...], and braces {...} are ignored.
        


        
Syntactic Grammar in ISO/IEC 14977 EBNF

        
(* Unfish Formal Language Grammar *)

Program
    = { Statement | TopLevelDecl } ;

TopLevelDecl
    = StructDecl
    | EnumDecl
    | ImportStmt
    | FunctionDecl ;

StructDecl
    = "struct" , Identifier , ":" , NEWLINE ,
      INDENT , { StructField | StructMethod } , DEDENT ;

StructField
    = Identifier , [ "=" , Expression ] , NEWLINE ;

StructMethod
    = ( "fn" | "function" ) , Identifier , "(" , [ ParameterList ] , ")" , ":" , Block ;

EnumDecl
    = "enum" , Identifier , ":" , NEWLINE ,
      INDENT , { VariantDecl } , DEDENT ;

VariantDecl
    = Identifier , [ "(" , IdentifierList , ")" ] , NEWLINE ;

ImportStmt
    = "import" , Identifier , [ "as" , Identifier ] , NEWLINE ;

FunctionDecl
    = ( "fn" | "function" ) , Identifier , "(" , [ ParameterList ] , ")" , ":" , Block ;

ParameterList
    = Parameter , { "," , Parameter } ;

Parameter
    = Identifier , [ "=" , Expression ] ;

Block
    = NEWLINE , INDENT , { Statement } , DEDENT ;

Statement
    = LetDecl
    | Assignment
    | IfStmt
    | WhileStmt
    | ForStmt
    | MatchStmt
    | TryCatchStmt
    | SayStmt
    | ReturnStmt
    | BreakStmt
    | ContinueStmt
    | ExpressionStmt ;

LetDecl
    = "let" , ( Identifier | DestructuringPattern ) , [ "=" , Expression ] , NEWLINE ;

DestructuringPattern
    = "[" , [ PatternList ] , "]"
    | "{" , [ MapPatternList ] , "}" ;

Assignment
    = PrimaryExpr , ( "=" | "+=" | "-=" | "*=" | "/=" | "%=" ) , Expression , NEWLINE ;

IfStmt
    = "if" , Expression , ":" , Block ,
      { "elif" , Expression , ":" , Block } ,
      [ "else" , ":" , Block ] ;

WhileStmt
    = "while" , Expression , ":" , Block ;

ForStmt
    = "for" , Identifier , "in" , Expression , ":" , Block ;

MatchStmt
    = "match" , Expression , ":" , NEWLINE ,
      INDENT , { MatchArm } , DEDENT ;

MatchArm
    = "when" , Pattern , ":" , Block ;

Pattern
    = Identifier , [ "(" , PatternList , ")" ]
    | Literal
    | "_" ;

TryCatchStmt
    = "try" , ":" , Block ,
      "catch" , [ Identifier ] , ":" , Block ,
      [ "finally" , ":" , Block ] ;

SayStmt
    = "say" , Expression , NEWLINE ;

ReturnStmt
    = "return" , [ Expression ] , NEWLINE ;

ExpressionStmt
    = Expression , NEWLINE ;

(* Expression Hierarchy with Pratt Precedence Climbing *)

Expression
    = PipelineExpr ;

PipelineExpr
    = LogicalOrExpr , { "|>" , ( Identifier | CallExpr ) } ;

LogicalOrExpr
    = LogicalAndExpr , { ( "or" | "||" ) , LogicalAndExpr } ;

LogicalAndExpr
    = EqualityExpr , { ( "and" | "&&" ) , EqualityExpr } ;

EqualityExpr
    = RelationalExpr , { ( "==" | "!=" ) , RelationalExpr } ;

RelationalExpr
    = AdditiveExpr , { ( "<" | "<=" | ">" | ">=" ) , AdditiveExpr } ;

AdditiveExpr
    = MultiplicativeExpr , { ( "+" | "-" ) , MultiplicativeExpr } ;

MultiplicativeExpr
    = UnaryExpr , { ( "*" | "/" | "%" ) , UnaryExpr } ;

UnaryExpr
    = ( "-" | "!" | "not" ) , UnaryExpr
    | PrimaryExpr ;

PrimaryExpr
    = Literal
    | Identifier
    | LambdaExpr
    | "(" , Expression , ")"
    | ArrayLiteral
    | MapLiteral
    | PrimaryExpr , "(" , [ ArgumentList ] , ")"
    | PrimaryExpr , "[" , Expression , "]"
    | PrimaryExpr , "." , Identifier ;

LambdaExpr
    = ( "fn" | "function" ) , "(" , [ ParameterList ] , ")" , ":" , ( Expression | Block ) ;

ArrayLiteral
    = "[" , [ Expression , { "," , Expression } ] , "]" ;

MapLiteral
    = "{" , [ MapEntry , { "," , MapEntry } ] , "}" ;

MapEntry
    = ( StringLiteral | Identifier ) , ":" , Expression ;

⚙️ 57-Opcode Stack VM ISA Specification

        

          The Unfish Virtual Machine is a bytecode-interpreted stack machine implemented in pure ANSI C99. It evaluates 16-byte tagged values (UfValue) on an operand stack with dynamic bounds checking, lexical upvalues for closures, and frame-pointer call unwinding.
        


        
Binary Bytecode Container (.ufc Format)

        
Precompiled Unfish bytecode modules are structured in a binary format with fixed endianness:

        
+-----------------------------------------------------------------------+
| Magic Bytes: 0x55 0x46 0x56 0x4D ("UFVM")               (4 bytes)     |
+-----------------------------------------------------------------------+
| Version: Major (u8), Minor (u8), Patch (u8), Flags (u8) (4 bytes)     |
+-----------------------------------------------------------------------+
| Constant Pool Count: uint32_t                           (4 bytes)     |
+-----------------------------------------------------------------------+
| Constant Pool Entries: [Type: u8, Length: u32, Data: bytes ...]       |
+-----------------------------------------------------------------------+
| Bytecode Instruction Stream: [Opcode: u8, Operands: var-length ...]   |
+-----------------------------------------------------------------------+
| Source Line Debug Mapping Table: [Offset: u32, Line: u32 ...]         |
+-----------------------------------------------------------------------+

🚀 256-Register VM Architecture (RegVM)

        

          While the stack VM provides a compact bytecode footprint, real-world interpretation often incurs high memory traffic due to constant operand pushes and pops. The Unfish RegVM provides a high-throughput alternative based on a 3-address register architecture.
        


        

          Run RegVM: Execute any Unfish script directly using the Register VM by passing the --regvm flag:
          
unfish run --regvm script.uf
        


        
32-Bit Fixed Instruction Formats

        
All RegVM instructions are encoded as 32-bit words for optimal cache line alignment and CPU branch prediction:

        
Format ABC:  [ Opcode : 8b ] [ Reg A : 8b ] [ Reg B : 8b ] [ Reg C : 8b ]
Format ABx:  [ Opcode : 8b ] [ Reg A : 8b ] [ Constant/Offset Bx : 16b  ]
Format AsBx: [ Opcode : 8b ] [ Reg A : 8b ] [ Signed Offset sBx : 16b   ]


        
Stack VM vs. Register VM Trace Comparison

        
Consider the arithmetic statement: let result = (a * b) + (c - d)


        

          

            
Stack VM (6 Instructions)

            
OP_LOAD_LOCAL  0    # push a
OP_LOAD_LOCAL  1    # push b
OP_MUL              # pop b, pop a, push (a*b)
OP_LOAD_LOCAL  2    # push c
OP_LOAD_LOCAL  3    # push d
OP_SUB              # pop d, pop c, push (c-d)
OP_ADD              # pop (c-d), pop (a*b), push sum
OP_STORE_LOCAL 4    # store into result

          

          

            
Register VM (3 Instructions)

            
ROP_MUL  r4, r0, r1   # r4 = r0 * r1
ROP_SUB  r5, r2, r3   # r5 = r2 - r3
ROP_ADD  r4, r4, r5   # r4 = r4 + r5 (result)

          

        


        
Register Windowing & Calling Convention

        

          Each execution fiber maintains an activation register file of 256 registers. Upon invoking a function:
        

        
    
          
  • The caller places function arguments into consecutive registers starting at r(A+1).
    • 
          
  • The call instruction ROP_CALL A, B, C shifts the callee's frame base pointer to rA.
    • 
          
  • Inside the callee, argument values map directly to r0, r1, ... without copying.
    • 
          
  • Upon return, ROP_RETURN A transfers the return value back to destination register rC.
    • 
        

      


      
      

        
🔌 C99 AOT & Host Embedding Guide

        

          Unfish was engineered from line 1 to be embeddable into host C and C++ applications. The entire runtime is self-contained in unfish_runtime.h with zero external dependencies beyond standard C99.
        


        
Compiling to Native Standalone C99

        
You can transpile any Unfish source code directly into portable ANSI C99 source code:

        
# Transpile Unfish to C99 source file
unfish emit-c -o game_logic.c game_logic.uf

# Compile directly to a native binary using GCC or Clang
gcc -O3 game_logic.c -lm -o game_logic


        
Embedding Unfish in a C Host Application

        
Here is a complete, working host program that embeds Unfish, executes scripts, and registers a custom C function callable from Unfish:


        
/* host_application.c — Embedding Unfish in C */
#include "unfish_runtime.h"

/* Custom native function exposed to Unfish */
static UfVal native_ocean_depth(void* env, size_t argc, UfVal* args) {
    (void)env;
    double trench_id = (argc > 0 && uf_is_num(args[0])) ? uf_to_num(args[0]) : 0.0;
    printf("[Host C] Fetching bathymetric depth for trench #%.0f\n", trench_id);
    return uf_num(10994.0); /* Mariana Trench depth in meters */
}

int main(void) {
    printf("=== Starting Host Application with Embedded Unfish ===\n");

    /* 1. Initialize host runtime context */
    UfRuntime* rt = uf_runtime_new();

    /* 2. Register native C function into Unfish globals */
    uf_register_native(rt, "get_ocean_depth", native_ocean_depth);

    /* 3. Execute Unfish script from string */
    const char* script =
        "let depth = get_ocean_depth(1)\n"
        "say f'Submersible reached depth: {depth} meters!'\n"
        "let pressure = (depth / 10.0) + 1.0\n"
        "say f'Ambient pressure: {round(pressure)} atmospheres'\n";

    UfValue result = uf_eval_string(rt, script);

    /* 4. Cleanup runtime memory */
    uf_runtime_free(rt);
    printf("=== Unfish runtime cleanly terminated ===\n");
    return 0;
}


        
Bare-Metal & Microcontroller Deployment (-DUF_EMBEDDED)

        

          For microcontrollers (ARM Cortex-M, ESP32, RISC-V) without standard POSIX memory allocators, compile with -DUF_EMBEDDED. Unfish replaces dynamic malloc with a fixed static buffer and routes text output to a custom UART transmitter:
        

        
/* Direct serial output to UART hardware */
void uart_send_char(char c) {
    while (!(USART1->SR & USART_SR_TXE));
    USART1->DR = c;
}

int main(void) {
    /* Route Unfish say statements to hardware UART */
    uf_set_putchar(uart_send_char);

    /* Runs fully within 256 KB static RAM buffer */
    uf_run_compiled_embedded();
    while (1);
}
